cmake_minimum_required(VERSION 3.16)
project(ConsoleViewCompatibility VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets Svg)
set(CONSOLE_VIEW_SOURCES src/main.cpp src/device_identity.cpp src/audio_monitor.cpp assets/resources.qrc)
if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "Oldest supported Intel macOS" FORCE)
    list(APPEND CONSOLE_VIEW_SOURCES src/mac_camera_bridge.mm)
    configure_file(../icon/PS4View.icns "${CMAKE_CURRENT_BINARY_DIR}/AppIcon.icns" COPYONLY)
    list(APPEND CONSOLE_VIEW_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/AppIcon.icns")
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/AppIcon.icns" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
elseif(WIN32)
    enable_language(RC)
    configure_file(assets/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    list(APPEND CONSOLE_VIEW_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
endif()
add_executable(consoleview WIN32 MACOSX_BUNDLE ${CONSOLE_VIEW_SOURCES})
target_compile_definitions(consoleview PRIVATE CONSOLE_VIEW_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(consoleview PRIVATE /utf-8)
    target_link_options(consoleview PRIVATE /SUBSYSTEM:WINDOWS,10.00)
endif()
target_link_libraries(consoleview PRIVATE Qt5::Widgets Qt5::Multimedia Qt5::MultimediaWidgets Qt5::Svg)
if(WIN32)
    target_link_libraries(consoleview PRIVATE setupapi cfgmgr32)
endif()
if(APPLE)
    target_link_libraries(consoleview PRIVATE "-framework AVFoundation" "-framework Foundation")
    set_target_properties(consoleview PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Console View"
        MACOSX_BUNDLE_GUI_IDENTIFIER "local.ismail.consoleview.compatibility"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_BUNDLE_VERSION "1"
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
endif()
install(TARGETS consoleview BUNDLE DESTINATION . RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    add_executable(consoleview-policy-tests tests/capture_policy_tests.cpp)
    target_include_directories(consoleview-policy-tests PRIVATE src)
    add_test(NAME capture-policy COMMAND consoleview-policy-tests)
    add_executable(consoleview-audio-tests tests/audio_ring_tests.cpp src/audio_monitor.cpp)
    target_include_directories(consoleview-audio-tests PRIVATE src)
    target_link_libraries(consoleview-audio-tests PRIVATE Qt5::Multimedia)
    add_test(NAME audio-ring COMMAND consoleview-audio-tests)
    add_test(NAME startup-smoke COMMAND consoleview --smoke-test)
    set_tests_properties(startup-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
